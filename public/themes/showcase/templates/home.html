# clone & install
git clone https://github.com/unifare/cfpress.git
cd cfpress
npm install

# create the two account-scoped resources
npx wrangler d1 create cfpress
npx wrangler kv namespace create CACHE

# start the dev server (port 47913)
npm run dev

# deploy — migrations first, then deploy
npm run db:migrate
npm run deploy
